\chapter{Logaritmos}\label{sec-logaritmos}

\begin{atlasnota}

\end{atlasnota}

\begin{exercise}[]\protect\hypertarget{exr-log-1}{}\label{exr-log-1}

Calcule \(\log_{3} 81\) y \(\log_{1/2} 8\) sin calculadora.

\end{exercise}

\emph{Solución.}

Como \(3^{4} = 81\), \(\log_{3} 81 = 4\). Como \((1/2)^{-3} = 8\),
\(\log_{1/2} 8 = -3\).

\begin{exercise}[]\protect\hypertarget{exr-log-2}{}\label{exr-log-2}

Demuestre que \(\log_b (x / y) = \log_b x - \log_b y\) para
\(x, y > 0\).

\end{exercise}

\emph{Solución.}

Por el Teorema~\ref{thm-log-producto},
\(\log_b x = \log_b\bigl((x/y) \, y\bigr) = \log_b (x/y) + \log_b y\).
Basta despejar.

\begin{exercise}[]\protect\hypertarget{exr-log-3}{}\label{exr-log-3}

¿Cuántos años tarda en triplicarse un capital al \(8\) \% efectivo
anual? Dé el resultado con dos decimales.

\end{exercise}

\emph{Solución.}

Se busca \(t\) con \(1{,}08^{t} = 3\), es decir,
\(t = \ln 3 / \ln 1{,}08 \approx 14{,}27\) años, por el mismo
razonamiento que llevó a la Ecuación~\ref{eq-duplicacion}.

\end{atlasnota}

\begin{exercise}[]\protect\hypertarget{exr-ito-1}{}\label{exr-ito-1}

Use el Teorema~\ref{thm-ito} para calcular \(\mathrm{d}(W_t^{2})\) y
deduzca que \(W_t^{2} - t\) es una martingala.

\end{exercise}

\emph{Solución.}

Con \(f(t, x) = x^{2}\), \(\mu = 0\) y \(\sigma = 1\), la
Ecuación~\ref{eq-ito} da
\(\mathrm{d}(W_t^{2}) = \mathrm{d}t + 2 W_t \, \mathrm{d}W_t\). Por
tanto \(W_t^{2} - t = \int_0^t 2 W_s \, \mathrm{d}W_s\), que es una
martingala porque
\(\mathbb{E}\int_0^t W_s^{2} \, \mathrm{d}s = t^{2}/2 < \infty\).
